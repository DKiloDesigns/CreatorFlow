### **Performance Optimization**

```tsx
import { useDebounce, useVirtualScrolling, useLazyLoading } from '@/lib/utils/performance-optimization';

function OptimizedList() {
  const [searchQuery, setSearchQuery] = useState('');
  const debouncedQuery = useDebounce(searchQuery, 300);
  
  const { data, loading, hasMore, loadMore } = useLazyLoading(
    () => fetchData(debouncedQuery),
    [],
    20
  );

  const { visibleItems, totalHeight, offsetY, elementRef } = useVirtualScrolling({
    items: data,
    itemHeight: 60,
    containerHeight: 400,
  });

  return (
    <div ref={elementRef} style={{ height: 400, overflow: 'auto' }}>
      <div style={{ height: totalHeight, position: 'relative' }}>
        <div style={{ transform: `translateY(${offsetY}px)` }}>
          {visibleItems.map(item => (
            <div key={item.id} style={{ height: 60 }}>
              {item.content}
            </div>
          ))}
        </div>
      </div>
    </div>
  );
}
```

### **Mobile Responsiveness**

```tsx
import { useTouchGestures, useBreakpoint, useHapticFeedback } from '@/lib/utils/mobile-responsiveness';

function MobileComponent() {
  const { isMobile, isTablet, isDesktop } = useBreakpoint();
  const { lightVibrate, mediumVibrate } = useHapticFeedback();
  const { onTouchStart, onTouchMove, onTouchEnd, gesture } = useTouchGestures();

  const handleGesture = (gestureType: string) => {
    if (gestureType === 'swipeLeft') {
      lightVibrate();
      // Handle swipe left
    } else if (gestureType === 'tap') {
      mediumVibrate();
      // Handle tap
    }
  };

  useEffect(() => {
    if (gesture) {
      handleGesture(gesture);
    }
  }, [gesture]);

  return (
    <div
      onTouchStart={onTouchStart}
      onTouchMove={onTouchMove}
      onTouchEnd={onTouchEnd}
      style={{
        padding: isMobile ? 16 : 24,
        fontSize: isMobile ? 14 : 16,
      }}
    >
      Mobile responsive content
    </div>
  );
}
```

```tsx
import { motion } from 'framer-motion';

function AnimatedComponent() {
  const { isHovered, hoverVariants, handleMouseEnter, handleMouseLeave } = useHoverAnimation();

  return (
    <motion.div

      animate={controls}

      onMouseLeave={handleMouseLeave}

      <motion.div variants={hoverVariants}>
        Animated content
      </motion.div>
    </motion.div>
  );
}
```
